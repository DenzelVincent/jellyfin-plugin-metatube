```python
def load_manifest() -> list:
    """Published manifest, or the built-in skeleton when it is unreachable.

    A fork's first publish has no dist branch yet, and a network hiccup must
    not abort a release that already built successfully.
    """
    try:
        with urlopen(MANIFEST_URL.format(repository=REPOSITORY), timeout=30) as f:
            return json.load(f)
    except (URLError, OSError, ValueError) as e:
        # HTTPError 是 URLError 的子类；JSON 解析失败走 ValueError。
        print(f'warning: published manifest unavailable, using skeleton: {e}',
              file=sys.stderr)
        return copy.deepcopy(SKELETON)
```

新建 `tmp/manifest_merge_check.py`（临时脚本，按约定放 `tmp/`，不提交）。它 monkeypatch 掉三处网络出口，然后真跑 `manifest.main()`：

```python
import io
import json
import os
import sys

os.environ['GITHUB_REPOSITORY'] = 'DenzelVincent/jellyfin-plugin-metatube'
sys.path.insert(0, os.path.join(os.path.dirname(__file__), '..', 'scripts'))
import manifest as m


class FakeResponse(io.BytesIO):
    def __enter__(self):
        return self

    def __exit__(self, *args):
        self.close()


PUBLISHED = json.dumps([{
    'guid': '01cc53ec-c415-4108-bbd4-a684a9801a32',
    'name': 'MetaTube',
    'description': 'old',
    'overview': 'old',
    'owner': 'old',
    'category': 'Metadata',
    'imageUrl': 'https://example.com/old.png',
    'versions': [
        {'checksum': 'KEEP-ME', 'changelog': 'old log', 'targetAbi': '10.11.0.0',
         'sourceUrl': 'https://github.com/whatever/old.zip',
         'timestamp': '2020-01-01T00:00:00Z', 'version': '2020.101.100.0'},
        {'checksum': 'DROP-ME', 'changelog': 'gone', 'targetAbi': '10.11.0.0',
         'sourceUrl': 'https://github.com/whatever/gone.zip',
         'timestamp': '2019-01-01T00:00:00Z', 'version': '2019.101.100.0'},
    ],
}]).encode()

m.urlopen = lambda *a, **k: FakeResponse(PUBLISHED)
m.md5sum_url = lambda url: 'REBUILT'
m.get_releases = lambda: [
    {'version': '2020.101.100.0', 'asset': 'x', 'timestamp': '2020-01-01T00:00:00Z'},
    {'version': '2021.101.100.0', 'asset': 'x', 'timestamp': '2021-01-01T00:00:00Z'},
]

work = os.path.dirname(os.path.abspath(__file__))
zip_name = 'Jellyfin.MetaTube@v2026.1103.1.0.zip'
open(os.path.join(work, zip_name), 'wb').write(b'plugin')
cwd = os.getcwd()
os.chdir(work)
sys.argv = ['manifest.py', zip_name]
m.main()
os.chdir(cwd)

entry = json.load(open(os.path.join(work, 'manifest.json')))[0]
old = [v for v in entry['versions'] if v['version'] == '2020.101.100.0'][0]
rebuilt = [v for v in entry['versions'] if v['version'] == '2021.101.100.0'][0]
current = entry['versions'][0]

assert entry['imageUrl'].startswith('https://raw.githubusercontent.com/DenzelVincent/'), entry['imageUrl']

assert old['timestamp'] == '2020-01-01T00:00:00Z', old

                           '/releases/download/v2020.101.100.0/Jellyfin.MetaTube@v2020.101.100.0.zip'), old

assert not any(v['version'] == '2019.101.100.0' for v in entry['versions']), 'release 已不存在的条目应被丢弃'
assert current['version'] == '2026.1103.1.0' and current['targetAbi'] == '10.11.0.0', current
print('OK: 合并语义保持（老条目不动、缺失条目重建、URL 全部指向本仓库）')
```

```csharp
namespace Jellyfin.Plugin.MetaTube.Helpers;

/// <summary>
/// 演员合集的纯计算：把「影片 → 演员」折算成「演员 → 影片」，再与库里现有合集算差集。
/// 这里不引用任何服务端类型，因此能脱离 Jellyfin 进程单测；Jellyfin 大版本改接口时
/// 也不用动这个文件（设计 5.3-8 把服务端调用全部关在扩展与任务里）。
/// </summary>
public static class ActorCollectionSync
{
    /// <summary>一条「某影片里的某演员」。Provider/ActorId 允许为空串——演员没刮到 pid 是常态。</summary>
    public readonly record struct ActorRef(Guid MovieId, string Name, string Provider, string ActorId);

    /// <summary>一位演员在本次扫描里的期望合集形态。</summary>
    public sealed class ActorGroup
    {
        public ActorGroup(string key, string displayName)
        {
            Key = key;
            DisplayName = displayName;
        }

        /// <summary>分组键：显示名 Trim + 小写。</summary>
        public string Key { get; }

        /// <summary>合集名，原样取演员显示名（设计 4.5：不加前后缀、不做模板）。</summary>
        public string DisplayName { get; }

        /// <summary>组内第一个非空 pid，只当合集封面来源，不参与分组。</summary>
        public string Provider { get; set; } = string.Empty;

        public string ActorId { get; set; } = string.Empty;

        public HashSet<Guid> MovieIds { get; } = new();
    }

    /// <summary>库里已有 BoxSet 的比对视图。</summary>
    public sealed class ExistingCollection
    {
        public Guid Id { get; set; }

        public string Name { get; set; } = string.Empty;

        public string Provider { get; set; } = string.Empty;

        public string ActorId { get; set; } = string.Empty;

        public HashSet<Guid> MemberIds { get; set; } = new();
    }

    public enum SyncAction
    {
        None,
        Create,
        Update
    }

    /// <summary>对一位演员的同步决定。</summary>
    public sealed class SyncPlan
    {
        public SyncAction Action { get; init; }

        public string Name { get; init; } = string.Empty;

        public string Provider { get; init; } = string.Empty;

        public string ActorId { get; init; } = string.Empty;

        /// <summary>Create 时为 null，Update 时为命中的合集 id。</summary>
        public Guid? CollectionId { get; init; }

        public IReadOnlyList<Guid> AddItemIds { get; init; } = Array.Empty<Guid>();

        public IReadOnlyList<Guid> RemoveItemIds { get; init; } = Array.Empty<Guid>();

        /// <summary>同名命中但合集上没有 pid，需要补写，供 CollectionImageProvider 解析封面。</summary>
        public bool NeedsProviderId { get; init; }

        /// <summary>非空表示本轮跳过这位演员的原因（只由最小片数阈值门控产生）。</summary>
        public string SkipReason { get; init; } = string.Empty;
    }

    public static string GroupKey(string name)
    {
        return (name ?? string.Empty).Trim().ToLowerInvariant();
    }

    public static Dictionary<string, ActorGroup> Build(IEnumerable<ActorRef> refs)
    {
        var groups = new Dictionary<string, ActorGroup>();
        foreach (var r in refs)
        {
            var key = GroupKey(r.Name);
            // 演员替换表把演员置空时库里留下的就是空显示名，不能给它建合集。
            if (key.Length == 0)
                continue;

            if (!groups.TryGetValue(key, out var group))
                groups[key] = group = new ActorGroup(key, r.Name.Trim());

            group.MovieIds.Add(r.MovieId);
            // 键是显示名而不是 pid：同一演员在库里完全可能一部分影片带 pid、一部分不带
            // （MovieProvider.SetActorImageUrl 只在演员有照片时才 SetPid），
            // 按 pid 分组会把这位演员裂成两个合集。
            if (group.ActorId.Length == 0 && !string.IsNullOrEmpty(r.Provider) && !string.IsNullOrEmpty(r.ActorId))
            {
                group.Provider = r.Provider;
                group.ActorId = r.ActorId;
            }
        }

        return groups;
    }

    /// <summary>
    /// 合集/演员的 MetaTube pid 串，格式与 MovieProvider 写入的一致（provider:id，id 做 URL 转义），
    /// 这样合集上的 pid 能用 ProviderIdsExtensions.GetPid 原样解回来。
    /// </summary>
    public static string EncodePid(string provider, string actorId)
    {
        return new ProviderId
        {
            Provider = provider,
            Id = Uri.EscapeDataString(actorId ?? string.Empty)
        }.ToString();
    }

    /// <summary>
    /// 现有合集索引。10.11 的 ICollectionManager 没有按名查找（设计第 3 节反射实测），
    /// 所以一次查询的结果在这里建两份索引，不能一位演员一查。
    /// </summary>
    public sealed class Index
    {
        private readonly Dictionary<string, ExistingCollection> _byPid = new(StringComparer.OrdinalIgnoreCase);

        // 键已经是 GroupKey 后的小写名，用默认序数比较即可。
        private readonly Dictionary<string, ExistingCollection> _byName = new();

        public Index(IEnumerable<ExistingCollection> collections)
        {
            foreach (var c in collections)
            {
                if (!string.IsNullOrEmpty(c.ActorId))
                    _byPid.TryAdd(EncodePid(c.Provider, c.ActorId), c);
                _byName.TryAdd(GroupKey(c.Name), c);
            }
        }

        /// <summary>先按 pid（尊重用户改名，只对齐成员不改名），再按同名（复用用户手建的合集并补写 pid）。</summary>
        public ExistingCollection Find(ActorGroup group)
        {
            if (!string.IsNullOrEmpty(group.ActorId) &&
                _byPid.TryGetValue(EncodePid(group.Provider, group.ActorId), out var byPid))
                return byPid;

            return _byName.TryGetValue(group.Key, out var byName) ? byName : null;
        }
    }

    /// <summary>
    /// 幂等判定：成员集合相等且 pid 不缺 → 不写库（设计 4.3）。
    /// 阈值只门控创建（设计 D12），跌破阈值不回收、不删除已有合集。
    /// </summary>
    public static SyncPlan Plan(ActorGroup group, ExistingCollection existing, ICollection<Guid> scopedMovieIds,
        int minMovieCount)
    {
        // 非法阈值按 1 处理；Warn 由调用方按配置值只记一次。
        var threshold = minMovieCount < 1 ? 1 : minMovieCount;

        if (existing is null)
        {
            if (group.MovieIds.Count < threshold)
                return new SyncPlan
                {
                    Action = SyncAction.None,
                    Name = group.DisplayName,
                    Provider = group.Provider,
                    ActorId = group.ActorId,
                    SkipReason = $"{group.DisplayName}: {group.MovieIds.Count} movie(s) below minimum count {threshold}"
                };

            return new SyncPlan
            {
                Action = SyncAction.Create,
                Name = group.DisplayName,
                Provider = group.Provider,
                ActorId = group.ActorId,
                // 排序让日志与差集稳定，重复运行不因顺序差异产生写入。
                AddItemIds = group.MovieIds.OrderBy(id => id).ToList()
            };
        }

        var add = group.MovieIds.Except(existing.MemberIds).OrderBy(id => id).ToList();
        // 只移除本次扫描范围内的影片：合集里手工加入的其它条目（别的库、别的来源）一律不动，
        // 否则一次同步就能把用户整理好的合集吃干净（设计 4.2-4）。
        var remove = existing.MemberIds.Intersect(scopedMovieIds).Except(group.MovieIds).OrderBy(id => id).ToList();
        var needsProviderId = string.IsNullOrEmpty(existing.ActorId) && !string.IsNullOrEmpty(group.ActorId);

        return new SyncPlan
        {
            Action = add.Count == 0 && remove.Count == 0 && !needsProviderId
                ? SyncAction.None
                : SyncAction.Update,
            Name = group.DisplayName,
            Provider = group.Provider,
            ActorId = group.ActorId,
            CollectionId = existing.Id,
            AddItemIds = add,
            RemoveItemIds = remove,
            NeedsProviderId = needsProviderId
        };
    }
}
```

```csharp
                // ItemIdList 是字符串形式的 Guid（设计第 3 节反射实测）。
                ItemIdList = plan.AddItemIds.Select(id => id.ToString()).ToList(),
                ProviderIds = BuildProviderIds(plan)
            }).ConfigureAwait(false);

            _logger.Info("Create actor collection: {0} ({1} movies)", plan.Name, plan.AddItemIds.Count);

            // 「pid 缺失」的另一种（设计 4.5）：这位演员一条 pid 都没刮到，合集建得出来、封面解析不了。
            // 只在建立时记一次，避免每轮对齐都刷屏。
            if (string.IsNullOrEmpty(plan.ActorId))
                _logger.Warn("Created actor collection without actor pid: {0}", plan.Name);

            // 不把服务端对 ItemIdList 的处理当唯一依据：成员数对不上就补一次，本身幂等。
            // LinkedChildren 可能为 null，口径与 ReadCollections 一致。
            var children = box.LinkedChildren ?? Array.Empty<LinkedChild>();
            var missing = plan.AddItemIds.Where(id => !children.Any(c => c.ItemId == id)).ToList();
            if (missing.Count > 0)
                await _collectionManager.AddToCollectionAsync(box.Id, missing).ConfigureAwait(false);

            return plan.Action;
        }

        if (plan.CollectionId is null)
            return ActorCollectionSync.SyncAction.None;

        var collectionId = plan.CollectionId.Value;

        if (plan.AddItemIds.Count > 0)
        {
            await _collectionManager.AddToCollectionAsync(collectionId, plan.AddItemIds).ConfigureAwait(false);
            _logger.Info("Add {0} movie(s) to actor collection: {1}", plan.AddItemIds.Count, plan.Name);
        }

        if (plan.RemoveItemIds.Count > 0)
        {
            await _collectionManager.RemoveFromCollectionAsync(collectionId, plan.RemoveItemIds)
                .ConfigureAwait(false);
            _logger.Info("Remove {0} movie(s) from actor collection: {1}", plan.RemoveItemIds.Count, plan.Name);
        }

        // pid 写在成员对齐之后：EnsureCollectionPidAsync 是「读整个合集 → 改 ProviderIds → 回写」，
        // 排在前面时它的快照早于本轮增删，回写有覆盖掉刚改完的成员的风险；成员失败中断时也不会留下
        // 「pid 已写但其实没对齐」的状态。
        if (plan.NeedsProviderId)
            await _libraryManager.EnsureCollectionPidAsync(collectionId, plan.Provider, plan.ActorId, _logger,
                cancellationToken).ConfigureAwait(false);

        return plan.Action;
    }

    /// <summary>
    /// 新建合集时就把演员 pid 一并写入，封面才能由 CollectionImageProvider 解析。
    /// 没有 pid 的演员照样建合集，只是封面得等用户手动指定。
    /// </summary>
    private static Dictionary<string, string> BuildProviderIds(ActorCollectionSync.SyncPlan plan)
    {
        var providerIds = new Dictionary<string, string>();
        if (string.IsNullOrEmpty(plan.Provider) || string.IsNullOrEmpty(plan.ActorId))
            return providerIds;

        providerIds[Plugin.ProviderId] = ActorCollectionSync.EncodePid(plan.Provider, plan.ActorId);
        return providerIds;
    }
}

```

```csharp
    public CollectionImageProvider(ILogger<CollectionImageProvider> logger) : base(logger)

    }

    {
        var pid = item.GetPid(Plugin.ProviderId);
        if (string.IsNullOrWhiteSpace(pid.Id) || string.IsNullOrWhiteSpace(pid.Provider))
            return Enumerable.Empty<RemoteImageInfo>();

        try
        {

            if (actorInfo?.Images?.Any() != true)
                return Enumerable.Empty<RemoteImageInfo>();

            return actorInfo.Images.Select(image => new RemoteImageInfo

                ProviderName = Name,
                Type = ImageType.Primary,
                Url = ApiClient.GetPrimaryImageApiUrl(actorInfo.Provider, actorInfo.Id, image, 0.66, true)
            });

        catch (OperationCanceledException)

            // 用户取消或服务端超时下发取消时必须原样抛出，否则这一次找图会被当成「正常但没图」。

            Logger.Warn("Fetch actor images for collection failed: {0} [{1}] {2}", item.Name,

            return Enumerable.Empty<RemoteImageInfo>();
        }
    }
}
```

```csharp
using Jellyfin.Plugin.MetaTube.Configuration;

namespace Jellyfin.Plugin.MetaTube.Helpers;

/// <summary>
/// 系列名译文表：整键查表 + 引擎新译结果回写持久化（设计 D14 / 4.6）。
/// 合集靠 CollectionName 按名字匹配，同一系列两次刮削拿到不同译文就会在库里裂成两个合集，
/// 所以缓存必须以「持久化后的 raw 串」为准，内存缓存挡不住重启后引擎换译法。
/// 只允许整键匹配：SubstitutionTable.Substitute 是子串批量替换，系列名互相包含时会误伤。
/// </summary>
public static class SeriesTranslationStore
{
    private static readonly object Sync = new();

    // SubstitutionTable 的构造函数是私有的，只能走 Parse 建实例。
    private static SubstitutionTable _table = SubstitutionTable.Parse(string.Empty);

    private static string _raw = string.Empty;

    /// <summary>纯查表，便于单测。命中即返回 true；表值为空串或 null 表示「保留原文」的显式意图。</summary>
    public static bool Lookup(string raw, string source, out string target)
    {
        target = null;
        if (string.IsNullOrWhiteSpace(raw) || string.IsNullOrWhiteSpace(source))
            return false;

        // 键要先裁再查：表里的键是 Parse 裁过的（`SubstitutionTable.cs:18`），
        // 拿没裁的原文去查，带空白的系列名永远 miss。
        return SubstitutionTable.Parse(raw).TryGetValue(source.Trim(), out target);
    }

    /// <summary>纯追加/替换一行，幂等：键值都相同时原样返回入参。含换行的键或值直接拒绝入表。</summary>
    public static string Append(string raw, string source, string target)
    {
        source = source?.Trim();
        target = target?.Trim();

        // 表是按行存的：Parse 逐行读、ToString 用 \n 拼。键或值里带换行就会写出多余的行，
        // 那行会被下次 Parse 读成「没有等号」的键——等于凭空多出一条「保留原文」的记录，只能手改 config.xml 才清得掉。
        if (string.IsNullOrEmpty(source) || source.Contains('\n') || source.Contains('\r') ||
            target != null && (target.Contains('\n') || target.Contains('\r')))
            return raw ?? string.Empty;

        var table = SubstitutionTable.Parse(raw);
        // 表值用 OrdinalIgnoreCase，与 SubstitutionTable 的键比较器同口径：键按大小写不敏感命中、
        // 值却按敏感比较，引擎这次给 "Abp" 下次给 "ABP" 就会来回改写同一行，合集名跟着裂（D14 要堵的正是这个）。
        if (table.TryGetValue(source, out var existing) && string.Equals(existing, target, StringComparison.OrdinalIgnoreCase))
            return raw ?? string.Empty;

        table[source] = target;
        return table.ToString();
    }

    /// <summary>
    /// 读当前配置里的译文表。缓存重建靠「当前 raw 串 ≠ 建表时的 raw 串」判断，
    /// 在锁内完成，这样一个缓存同时覆盖用户在配置页的编辑与刮削期的自动回写，
    /// 不必每张影片重解析文本。
    /// </summary>
    public static bool TryGetValue(string source, out string target)
    {
        target = null;
        if (string.IsNullOrWhiteSpace(source))
            return false;

        lock (Sync)
        {
            var raw = Plugin.Instance.Configuration.RawSeriesTranslationTable ?? string.Empty;
            if (raw != _raw)
            {
                _table = SubstitutionTable.Parse(raw);
                _raw = raw;
            }

            return _table.TryGetValue(source.Trim(), out target);
        }
    }

#if !__EMBY__
    /// <summary>
    /// 把引擎新译的结果写回配置，下一轮刮削直接命中、不再调用引擎。
    /// 返回值是「真正生效的译文」：入库名必须与表里那行一致，否则同系列的多部影片会裂成两个合集（设计 D14）。
    /// 刮削是并发的，「读当前 raw → 判断 → 追加一行 → 整体写回」必须在同一个锁里做完（设计 V8）。
    /// Emby 端不做自动回写：BasePluginSimpleUI 怎么从刮削线程持久化配置没验证过（设计 4.4）。
    /// </summary>
    public static string Persist(string source, string target, Action<string> log)
    {
        try
        {
            lock (Sync)
            {
                var config = Plugin.Instance.Configuration;
                var raw = config.RawSeriesTranslationTable ?? string.Empty;

                // 能走到这里说明查表时还没有这一行；现在有了就是另一个并发线程先写完，采用先到的译文。
                // 同一次全库刷新里，一个系列的十几部影片并发走到这条链路是常态而不是边角。
                // 表值为空或 null 同样不覆盖：那是用户「这个系列保留原文」的显式意图。
                if (Lookup(raw, source, out var existing))
                {
                    log?.Invoke($"Series translation row appeared meanwhile, reuse it: {source} => {existing}");
                    return existing;
                }

                var next = Append(raw, source, target);
                if (next != raw)
                {
                    config.RawSeriesTranslationTable = next;
                    Plugin.Instance.UpdateConfiguration(config);
                    log?.Invoke($"Persist series translation: {source} => {target}");
                }

                return target;
            }
        }
        catch (Exception e)
        {
            // 持久化失败只影响下次要不要再调一次引擎，不改变本次刮削结果。
            log?.Invoke($"Persist series translation failed: {e.Message}");
            return target;
        }
    }
#endif
}
```

```csharp
    /// 系列名单独一条链路：先查持久化对照表，命中就不再调引擎（含「表值为空 = 保留原文」）。
    /// 合集名、系列标签与 {series} 模板变量都取这一个值，所以译法必须每次都一样。
    /// </summary>
    private static async Task<string> TranslateSeriesAsync(string series, string to,
        CancellationToken cancellationToken, Action<string> log)
    {
        if (SeriesTranslationStore.TryGetValue(series, out var cached))
            return string.IsNullOrWhiteSpace(cached) ? series : cached;

        // 目标语言是中文且系列名已含汉字（iqqtv/airav 直接给中文系列名）时省一次调用。
        if (ShouldSkipTranslation(series, to))
            return series;

        var translated = await TranslateFieldSafeAsync(series, to, cancellationToken, log).ConfigureAwait(false);
        // 先裁再比对：不裁的话 " 甲 " != "甲" 会被当成一次成功翻译，把带空白的串当译文入库。
        translated = translated?.Trim();
        // 引擎失败时 TranslateFieldSafeAsync 原样返回，此时不写表，免得把原文当译文固化下来。
        if (string.IsNullOrWhiteSpace(translated) || translated == series)
            return series;

        // 译文要直接当合集名用，带换行就存不进按行存的译文表、也建不出正常合集；宁可保留原文。
        if (translated.Contains('\n') || translated.Contains('\r'))
        {
            log($"Translated series contains line breaks, keep original: {series}");
            return series;
        }

#if !__EMBY__
        // 生效译文可能不是引擎这次给的：同一系列的另一部影片可能已经先写了一行（Persist 先到先得）。
        var effective = SeriesTranslationStore.Persist(series, translated, log);
        return string.IsNullOrWhiteSpace(effective) ? series : effective;
#else
        return translated;
#endif
    }
```

属性化后 `get_jellyfin_version` 拿到的是字符串 `$(JellyfinSdkVersion)`，`Version()` 当场抛异常、CI 在生成 manifest 这步中断（设计第 3 节最后一条）。改成回查同一份 csproj 的属性默认值，**不能静默回落成写死的版本**，否则 `targetAbi` 会说谎：

```python
def resolve_version(root, version: str) -> str:
    """Resolve a property-ised package version like "$(JellyfinSdkVersion)".

    Never fall back to a hard-coded default: targetAbi must describe the build
    that actually happened.
    """
    if not version.startswith("$("):
        return version

    name = version[2:-1]
    for prop in root.iter(name):
        if prop.text and prop.text.strip():
            return prop.text.strip()

    raise Exception(f"Cannot resolve {version} for the Jellyfin package version")


def get_jellyfin_version(csproj: str) -> str:
    tree = ET.parse(csproj)
    root = tree.getroot()

    for pkg in root.iter("PackageReference"):
        if pkg.attrib.get("Include") in ("Jellyfin.Controller", "Jellyfin.Model"):
            return Version(resolve_version(root, pkg.attrib.get("Version"))).base_version

    raise Exception("Jellyfin version not found")
```

这是设计目标 7 的落地机制：以后升 Jellyfin 大版本前先跑一次，而不是升完才发现编不过。输入走 `env` 而不是直接插值进 `run`，避免 workflow_dispatch 参数变成 shell 注入点。

```yaml
name: Jellyfin 预览编译

# 只试编、不发布产物：升 Jellyfin 大版本前先跑一次，把破坏点提前暴露出来。
# 失败不等于仓库坏了——发布链路仍然按 csproj 里的默认版本走。
on:
  workflow_dispatch:
    inputs:
      sdk_version:
        description: '要试编的 Jellyfin SDK 版本'
        required: true
        default: '12.4.0'
        type: string
      target_framework:
        description: '与之配对的目标框架（12.x 只支持 net10.0）'
        required: true
        default: 'net10.0'
        type: string

jobs:
  preview:
    runs-on: ubuntu-latest
    env:
      SDK_VERSION: ${{ github.event.inputs.sdk_version }}
      TARGET_FRAMEWORK: ${{ github.event.inputs.target_framework }}
    steps:
      - uses: actions/checkout@v4

      # 6.0.x 是 Emby 目标（net6.0）需要的，删了会连带把 Emby 目标编挂。
      - name: Setup .NET
        uses: actions/setup-dotnet@v4
        with:
          dotnet-version: |
            6.0.x
            9.0.x
            10.0.x

      # 只编插件工程：测试工程硬编码 net9.0（`Jellyfin.Plugin.MetaTube.Tests.csproj:5`），
      # 裸跑会连它一起编，解决方案级报 NU1201，掩盖真正的破坏点。理由见 E2.4。
      - name: Build Jellyfin target
        run: dotnet build Jellyfin.Plugin.MetaTube/Jellyfin.Plugin.MetaTube.csproj --configuration Release -p:JellyfinSdkVersion="$SDK_VERSION" -p:JellyfinTargetFramework="$TARGET_FRAMEWORK"

      # Emby 目标不吃这两个参数，一起编一遍确认没被波及。
      - name: Build Emby target
        run: dotnet build --configuration Release.Emby
```
